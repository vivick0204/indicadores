/** * PAINEL DE ORÇAMENTOS – API da planilha (Google Apps Script) * ------------------------------------------------------------------ * O que este script faz: * 1. doGet -> entrega para a página HTML os dados das abas de orçamentos, * aprovados, contratos, agenda dos contratos e NFs de contratos, * em JSON (somente leitura). * 2. doPost -> grava o que foi alterado na página e registra * quem/quando/o que mudou na aba de log: * - acao "atualizar" : Valor, Status, Data Aprovação ou Observações de um orçamento * - acao "nf_salvar" : inclui ou corrige uma linha (NF) na aba "Emissão de NFs" * - acao "nf_excluir" : apaga uma NF lançada por engano * * VERSÃO 5 (23/11/2026): a aba "Emissão de NFs" virou uma lista, uma NF por linha: * Contrato | Mês de referência | Ano | Nº NF | Data de emissão | Valor | Observação | Lançado por | Lançado em * A grade contrato × mês fica na aba "Resumo NFs" (fórmulas). Datas gravadas ao meio-dia, * para não mudarem de dia por causa do fuso horário da planilha. * VERSÃO 4 (23/11/2026): NFs separadas por ";" na mesma célula; a página avisa quando o * script publicado é de versão anterior ou quando a chave é só de leitura. * VERSÃO 3 (23/11/2026): as NFs ficam na aba "Emissão de NFs" (uma linha por contrato, * uma coluna por mês). O endereço /exec continua o mesmo. * * Como instalar (resumo – o passo a passo completo está no LEIAME.md): * a) Na planilha: Extensões > Apps Script. Apague o conteúdo e cole este arquivo. * b) Rode a função configurarChaves() uma vez e autorize. Copie as duas chaves do log. * c) Implantar > Nova implantação > App da Web * Executar como: Eu Quem pode acessar: Qualquer pessoa * d) Copie o endereço que termina em /exec e informe na página. * * Segurança: sem a chave correta o script não devolve nem grava nada. * A chave de leitura só lê. A chave de edição lê e grava. */ const VERSAO_SCRIPT = 5; const CONFIG = { // Deixe vazio para usar a planilha onde o script está colado. // Preencha com o ID da planilha só se o script for criado fora dela. PLANILHA_ID: '', ABA_ORCAMENTOS: 'RELAÇÃO ORÇAMENTOS - 2026', ABA_APROVADOS: 'Aprovados 2026', ABA_CONTRATOS: 'Contratos', ABA_AGENDA: 'Controle agendamento contratos ativos', ABA_EMISSAO: 'Emissão de NFs', CAB_EMISSAO: ['Contrato', 'Mês de referência', 'Ano', 'Nº NF', 'Data de emissão', 'Valor', 'Observação', 'Lançado por', 'Lançado em'], MESES: ['janeiro', 'fevereiro', 'março', 'abril', 'maio', 'junho', 'julho', 'agosto', 'setembro', 'outubro', 'novembro', 'dezembro'], ABA_LOG: 'Log alterações (página)', // Valores aceitos na coluna Status da aba de orçamentos STATUS_VALIDOS: ['APROVADO', 'NÃO APROVADA', 'AGUARDANDO APROVAÇÃO', 'CANCELADA'], // Campos que a página pode alterar -> título da coluna na aba de orçamentos CAMPOS_EDITAVEIS: { valor: 'Valor', status: 'Status', dataAprovacao: 'Data Aprovação', obs: 'OBSERVAÇÕES FOLLOW UP' }, COLUNA_OS: 'OS', COLUNA_CLIENTE: 'Cliente' }; /* ============================ CHAVES ============================ */ /** Rode uma vez. Cria as chaves (se ainda não existirem) e mostra no log. */ function configurarChaves() { const p = PropertiesService.getScriptProperties(); if (!p.getProperty('CHAVE_LEITURA')) p.setProperty('CHAVE_LEITURA', gerarChave_()); if (!p.getProperty('CHAVE_EDICAO')) p.setProperty('CHAVE_EDICAO', gerarChave_()); mostrarChaves(); } /** Mostra as chaves atuais em Execuções > Registros. */ function mostrarChaves() { const p = PropertiesService.getScriptProperties(); Logger.log('CHAVE DE LEITURA: ' + p.getProperty('CHAVE_LEITURA')); Logger.log('CHAVE DE EDIÇÃO : ' + p.getProperty('CHAVE_EDICAO')); } /** Troca as duas chaves (use se alguma vazar). As antigas param de funcionar na hora. */ function trocarChaves() { const p = PropertiesService.getScriptProperties(); p.setProperty('CHAVE_LEITURA', gerarChave_()); p.setProperty('CHAVE_EDICAO', gerarChave_()); mostrarChaves(); } function gerarChave_() { return (Utilities.getUuid() + Utilities.getUuid()).replace(/-/g, '').slice(0, 40); } /** Devolve 'edicao', 'leitura' ou '' conforme a chave recebida. */ function nivel_(chave) { const p = PropertiesService.getScriptProperties(); const k = String(chave || ''); if (!k) return ''; if (k === p.getProperty('CHAVE_EDICAO')) return 'edicao'; if (k === p.getProperty('CHAVE_LEITURA')) return 'leitura'; return ''; } /* ============================ LEITURA ============================ */ function doGet(e) { try { const nivel = nivel_(e && e.parameter && e.parameter.chave); if (!nivel) return json_({ ok: false, erro: 'Chave inválida ou não informada.' }); const ss = planilha_(); return json_({ ok: true, atualizado: Utilities.formatDate(new Date(), ss.getSpreadsheetTimeZone(), "dd/MM/yyyy HH:mm:ss"), planilha: ss.getName(), podeEditar: nivel === 'edicao', versao: VERSAO_SCRIPT, statusValidos: CONFIG.STATUS_VALIDOS, orc: lerAba_(ss, CONFIG.ABA_ORCAMENTOS, [CONFIG.COLUNA_OS, CONFIG.COLUNA_CLIENTE]), apr: lerAba_(ss, CONFIG.ABA_APROVADOS, ['OS', 'Cliente']), contratos: lerAba_(ss, CONFIG.ABA_CONTRATOS, ['Cliente']), agenda: ss.getSheetByName(CONFIG.ABA_AGENDA) ? lerAba_(ss, CONFIG.ABA_AGENDA, ['Cliente']) : { cab: [], linhas: [] }, emissao: ss.getSheetByName(CONFIG.ABA_EMISSAO) ? lerAba_(ss, CONFIG.ABA_EMISSAO, ['Contrato', 'Nº NF']) : null, fuso: ss.getSpreadsheetTimeZone() }); } catch (err) { return json_({ ok: false, erro: String(err && err.message || err) }); } } /** * Lê uma aba inteira como texto (do jeito que aparece na tela). * Devolve { cab: [títulos], linhas: [[nº da linha na planilha, célula1, célula2, ...], ...] }. * Só entram linhas em que pelo menos uma das colunas-chave está preenchida. */ function lerAba_(ss, nome, colunasChave) { const sh = ss.getSheetByName(nome); if (!sh) throw new Error('Aba não encontrada: "' + nome + '". Confira o nome em CONFIG.'); const nLin = sh.getLastRow(), nCol = sh.getLastColumn(); if (nLin < 1 || nCol < 1) return { cab: [], linhas: [] }; const v = sh.getRange(1, 1, nLin, nCol).getDisplayValues(); const cab = v[0].map(function (h) { return String(h).trim(); }); const idx = colunasChave.map(function (c) { return indice_(cab, c); }).filter(function (i) { return i >= 0; }); const linhas = []; for (let i = 1; i < v.length; i++) { const r = v[i]; let tem = false; for (let j = 0; j < idx.length; j++) if (String(r[idx[j]]).trim() !== '') { tem = true; break; } if (tem) linhas.push([i + 1].concat(r)); } return { cab: cab, linhas: linhas }; } /* ============================ GRAVAÇÃO ============================ */ function doPost(e) { let d; try { d = JSON.parse(e.postData.contents); } catch (err) { return json_({ ok: false, erro: 'Pedido inválido.' }); } if (nivel_(d.chave) !== 'edicao') return json_({ ok: false, erro: 'Esta chave não permite edição.' }); const lock = LockService.getScriptLock(); try { lock.waitLock(20000); if (d.acao === 'atualizar') return json_(atualizar_(d)); if (d.acao === 'nf_salvar') return json_(nfSalvar_(d)); if (d.acao === 'nf_excluir') return json_(nfExcluir_(d)); return json_({ ok: false, erro: 'Ação desconhecida.' }); } catch (err) { return json_({ ok: false, erro: String(err && err.message || err) }); } finally { try { lock.releaseLock(); } catch (x) {} } } /** * d = { linha: 123, os: '98764', campos: { valor: 1500.5, status: 'APROVADO', * dataAprovacao: '12/11/2026', obs: 'texto' }, quem: 'Vitória' } * Só os campos enviados são alterados. */ function atualizar_(d) { const ss = planilha_(); const sh = ss.getSheetByName(CONFIG.ABA_ORCAMENTOS); if (!sh) throw new Error('Aba não encontrada: ' + CONFIG.ABA_ORCAMENTOS); const cab = sh.getRange(1, 1, 1, sh.getLastColumn()).getDisplayValues()[0].map(function (h) { return String(h).trim(); }); const colOS = indice_(cab, CONFIG.COLUNA_OS) + 1; if (colOS < 1) throw new Error('Coluna "OS" não encontrada na aba de orçamentos.'); // 1) Localiza a linha. A página manda o nº da linha e a OS; se alguém inseriu ou // apagou linhas depois que a página carregou, a OS não bate e procuramos de novo. const os = String(d.os || '').trim(); if (!os) throw new Error('OS não informada.'); let linha = Number(d.linha); const ultima = sh.getLastRow(); const confere = linha >= 2 && linha <= ultima && String(sh.getRange(linha, colOS).getDisplayValue()).trim() === os; if (!confere) { const todas = sh.getRange(2, colOS, ultima - 1, 1).getDisplayValues(); const achadas = []; for (let i = 0; i < todas.length; i++) if (String(todas[i][0]).trim() === os) achadas.push(i + 2); if (achadas.length === 0) throw new Error('OS ' + os + ' não encontrada na planilha. Atualize a página.'); if (achadas.length > 1) throw new Error('OS ' + os + ' aparece em ' + achadas.length + ' linhas (' + achadas.join(', ') + '). Corrija direto na planilha.'); linha = achadas[0]; } // 2) Valida tudo antes de gravar qualquer coisa const campos = d.campos || {}; const gravar = []; Object.keys(campos).forEach(function (k) { const titulo = CONFIG.CAMPOS_EDITAVEIS[k]; if (!titulo) throw new Error('Campo não permitido: ' + k); const col = indice_(cab, titulo) + 1; if (col < 1) throw new Error('Coluna "' + titulo + '" não encontrada na aba de orçamentos.'); let valor = campos[k], formato = null; if (k === 'valor') { valor = numero_(valor); if (valor === null || valor < 0) throw new Error('Valor inválido.'); } else if (k === 'status') { valor = String(valor).trim().toUpperCase(); if (CONFIG.STATUS_VALIDOS.indexOf(valor) < 0) throw new Error('Status inválido: ' + valor); } else if (k === 'dataAprovacao') { valor = String(valor || '').trim(); if (valor !== '') { valor = data_(valor); if (!valor) throw new Error('Data inválida. Use dd/mm/aaaa.'); formato = 'dd/MM/yyyy'; } } else { valor = String(valor == null ? '' : valor).slice(0, 2000); } gravar.push({ campo: titulo, col: col, valor: valor, formato: formato }); }); if (!gravar.length) throw new Error('Nenhuma alteração enviada.'); // 3) Grava e registra no log const log = abaLog_(ss); const agora = new Date(); const quem = String(d.quem || '').slice(0, 80) || '(não informado)'; const cliente = (function () { const c = indice_(cab, CONFIG.COLUNA_CLIENTE) + 1; return c > 0 ? sh.getRange(linha, c).getDisplayValue() : ''; })(); const feitas = []; gravar.forEach(function (g) { const cel = sh.getRange(linha, g.col); const antes = cel.getDisplayValue(); if (g.valor === '') cel.clearContent(); else cel.setValue(g.valor); if (g.formato) cel.setNumberFormat(g.formato); SpreadsheetApp.flush(); const depois = cel.getDisplayValue(); log.appendRow([agora, quem, os, cliente, linha, g.campo, antes, depois]); feitas.push({ campo: g.campo, antes: antes, depois: depois }); }); return { ok: true, linha: linha, os: os, alteracoes: feitas }; } /* ============================ NFs DE CONTRATO ============================ */ /* * A página manda: * linha : linha da aba "Emissão de NFs" (0 = NF nova) * antes : como a linha estava quando a página carregou (as 7 primeiras colunas, separadas por "|"). * Se a linha mudou na planilha nesse meio-tempo, a gravação é recusada. * nf : { contrato, mes (1 a 12), ano, numero, data ('dd/mm/aaaa'), valor, obs } (só em nf_salvar) */ const COLS_ASSINATURA = 7; function nfSalvar_(d) { const ss = planilha_(), nf = d.nf || {}; const contrato = String(nf.contrato || '').trim(); const mes = Number(nf.mes), ano = Number(nf.ano); const numero = String(nf.numero || '').trim(); const valor = numero_(nf.valor); const data = data_(nf.data); const obs = String(nf.obs || '').slice(0, 300); if (contratosExistentes_(ss).indexOf(contrato) < 0) throw new Error('Contrato "' + contrato + '" não encontrado na aba ' + CONFIG.ABA_CONTRATOS + '.'); if (!(mes >= 1 && mes <= 12)) throw new Error('Mês de referência inválido.'); if (!(ano >= 2020 && ano <= 2040)) throw new Error('Ano inválido.'); if (!numero) throw new Error('Informe o número da NF.'); if (!data) throw new Error('Data de emissão inválida. Use dd/mm/aaaa.'); if (valor === null || valor <= 0) throw new Error('Valor da NF inválido.'); data.setHours(12); // meio-dia: o fuso da planilha não troca o dia const t = abaEmissao_(ss), linha = Number(d.linha) || 0; if (linha) conferirLinha_(t, linha, d.antes); // número repetido em outra linha (corrigir a própria linha sem trocar o número é permitido) const numAtual = linha ? String(t.v[linha - 1][t.col['Nº NF']]).trim() : ''; if (numero !== numAtual) { for (let i = 1; i < t.v.length; i++) { if (i + 1 !== linha && String(t.v[i][t.col['Nº NF']]).trim() === numero) { throw new Error('A NF nº ' + numero + ' já está lançada (linha ' + (i + 1) + ': ' + t.v[i][t.col['Contrato']] + ', ' + t.v[i][t.col['Mês de referência']] + ').'); } } } const quem = String(d.quem || '').slice(0, 80) || '(não informado)', agora = new Date(); const novos = {}; novos['Contrato'] = contrato; novos['Mês de referência'] = CONFIG.MESES[mes - 1]; novos['Ano'] = ano; novos['Nº NF'] = numero; novos['Data de emissão'] = data; novos['Valor'] = valor; novos['Observação'] = obs; novos['Lançado por'] = quem; novos['Lançado em'] = agora; const destino = linha || proximaLinha_(t); const antes = linha ? t.v[linha - 1].slice(0, COLS_ASSINATURA).join(' | ') : ''; Object.keys(novos).forEach(function (titulo) { const c = t.col[titulo]; if (c === undefined) return; const cel = t.sh.getRange(destino, c + 1); if (titulo === 'Nº NF' || titulo === 'Contrato' || titulo === 'Mês de referência' || titulo === 'Observação') cel.setNumberFormat('@'); if (titulo === 'Data de emissão') cel.setNumberFormat('dd/mm/yyyy'); if (titulo === 'Valor') cel.setNumberFormat('"R$" #,##0.00'); if (titulo === 'Lançado em') cel.setNumberFormat('dd/mm/yyyy hh:mm'); cel.setValue(novos[titulo]); }); SpreadsheetApp.flush(); const depois = t.sh.getRange(destino, 1, 1, COLS_ASSINATURA).getDisplayValues()[0].join(' | '); abaLog_(ss).appendRow([agora, quem, 'NF ' + numero, contrato, destino, linha ? 'NF corrigida' : 'NF incluída', antes, depois]); return { ok: true, linha: destino }; } /** d = { linha, antes } */ function nfExcluir_(d) { const ss = planilha_(), t = abaEmissao_(ss), linha = Number(d.linha); if (!(linha >= 2)) throw new Error('Linha inválida.'); conferirLinha_(t, linha, d.antes); const antes = t.v[linha - 1].slice(0, COLS_ASSINATURA).join(' | '); abaLog_(ss).appendRow([new Date(), String(d.quem || '').slice(0, 80) || '(não informado)', 'NF ' + t.v[linha - 1][t.col['Nº NF']], t.v[linha - 1][t.col['Contrato']], linha, 'NF excluída', antes, '']); t.sh.deleteRow(linha); return { ok: true }; } /** Lê a aba de NFs inteira e acha as colunas pelo título. */ function abaEmissao_(ss) { const sh = ss.getSheetByName(CONFIG.ABA_EMISSAO); if (!sh) throw new Error('Aba não encontrada: "' + CONFIG.ABA_EMISSAO + '".'); const v = sh.getRange(1, 1, Math.max(sh.getLastRow(), 1), Math.max(sh.getLastColumn(), CONFIG.CAB_EMISSAO.length)).getDisplayValues(); const cab = v[0].map(function (h) { return String(h).trim(); }), col = {}; CONFIG.CAB_EMISSAO.forEach(function (t) { const i = indice_(cab, t); if (i >= 0) col[t] = i; }); ['Contrato', 'Mês de referência', 'Ano', 'Nº NF', 'Data de emissão', 'Valor'].forEach(function (t) { if (col[t] === undefined) throw new Error('Coluna "' + t + '" não encontrada na aba ' + CONFIG.ABA_EMISSAO + '.'); }); return { sh: sh, v: v, col: col }; } /** Recusa se a linha não está mais como a página viu. */ function conferirLinha_(t, linha, antes) { if (linha > t.v.length) throw new Error('Essa NF não está mais na planilha. Clique em Atualizar agora.'); const atual = t.v[linha - 1].slice(0, COLS_ASSINATURA).map(function (x) { return String(x).trim(); }).join('|'); if (atual !== String(antes || '')) throw new Error('Essa linha foi alterada na planilha depois que a página carregou. Clique em Atualizar agora e tente de novo.'); } /** Primeira linha depois da última que tem Contrato ou Nº NF. */ function proximaLinha_(t) { let ultima = 1; for (let i = 1; i < t.v.length; i++) if (String(t.v[i][t.col['Contrato']]).trim() || String(t.v[i][t.col['Nº NF']]).trim()) ultima = i + 1; return ultima + 1; } function contratosExistentes_(ss) { const sh = ss.getSheetByName(CONFIG.ABA_CONTRATOS); if (!sh) throw new Error('Aba não encontrada: ' + CONFIG.ABA_CONTRATOS); const v = sh.getRange(1, 1, sh.getLastRow(), sh.getLastColumn()).getDisplayValues(); const c = indice_(v[0].map(function (h) { return String(h).trim(); }), 'Cliente'); return v.slice(1).map(function (r) { return String(r[c]).trim(); }).filter(String); } /* ============================ APOIO ============================ */ function planilha_() { return CONFIG.PLANILHA_ID ? SpreadsheetApp.openById(CONFIG.PLANILHA_ID) : SpreadsheetApp.getActiveSpreadsheet(); } function indice_(cab, titulo) { const t = String(titulo).trim().toLowerCase(); for (let i = 0; i < cab.length; i++) if (String(cab[i]).trim().toLowerCase() === t) return i; return -1; } /** Aceita 1500.5, "1500,50", "1.500,50" ou "R$ 1.500,50". */ function numero_(v) { if (typeof v === 'number') return isFinite(v) ? v : null; let s = String(v || '').replace(/R\$/g, '').replace(/\s/g, ''); if (s === '') return null; if (s.indexOf(',') >= 0) s = s.replace(/\./g, '').replace(',', '.'); const n = Number(s); return isFinite(n) ? n : null; } /** "12/11/2026" -> Date. Devolve null se a data não existir. */ function data_(s) { const m = /^(\d{1,2})\/(\d{1,2})\/(\d{4})$/.exec(String(s).trim()); if (!m) return null; const dt = new Date(Number(m[3]), Number(m[2]) - 1, Number(m[1]), 12); if (dt.getFullYear() !== Number(m[3]) || dt.getMonth() !== Number(m[2]) - 1 || dt.getDate() !== Number(m[1])) return null; return dt; } function abaLog_(ss) { let sh = ss.getSheetByName(CONFIG.ABA_LOG); if (!sh) { sh = ss.insertSheet(CONFIG.ABA_LOG); sh.appendRow(['Data/hora', 'Quem', 'OS', 'Cliente', 'Linha', 'Campo', 'Valor anterior', 'Valor novo']); sh.setFrozenRows(1); sh.getRange('A:A').setNumberFormat('dd/MM/yyyy HH:mm:ss'); } return sh; } function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }
